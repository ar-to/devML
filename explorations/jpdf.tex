
\text{typical min-max scaling or normalization function}
\\\\

x_{\text{scaled}} = \frac{x - x_{\min}}{x_{\max} - x_{\min}}


\\\\

x_{\text{scaled}} = a + \frac{(x - x_{\min})(b - a)}{x_{\max} - x_{\min}}


\\\\

\text{1D (single variable) Probability Density Function (PDF)}
\\\\
\text{non-negativity and integrate to 1 over domain}
\\\\

P(a \le X \le b) = \int_{a}^{b} f(x) \,dx

\\\\

\begin{align*}
1. & \quad f(x) \ge 0 \quad \text{for all } x \\
2. & \quad \int_{a}^{b} f(x) \,dx = 1
\end{align*}

\\\\

\text{2D (2 variables) Joint Probability Density Function (JPDF) & Double Integral}

\\\\

f(x_1, x_2) = \text{JPDF}

\\\\

P(a \leq x_1 \leq b \text{ and } c \leq x_2 \leq d) = \int_{c}^{d} \int_{a}^{b} f(x_1, x_2) \, dx_1 \, dx_2


\\\\

\int_{c}^{d} \int_{a}^{b} f(x_1, x_2) \, dx_1 \, dx_2 = 1


\\\\

\text{Assignment function}
\\\
p(x_1,x_2) = \frac{(x_1-1)x_2}{9}

\\\\

\text{Domain}

\\\\

x_1 \in [1, 4], \quad x_2 \in [0, 2]

\\\\

\text{solution/hypothesis}

\\\\

\int \int p(x_1,x_2) \, dx_1 \, dx_2 = 1
  
\\\\

\text{PROOF}

\\\\

\text{PDF of two continuous variables on 3D dimensions}

\\\\
f(x_1, x_2)

\\\\

\text{Get 9 denominator for normalizing constant; to scale JPDF to 1}

\\\\

\begin{aligned}
\text{C} &= \int_{0}^{2} \int_{1}^{4} (x_1 - 1)x_2 \, dx_1 \, dx_2 \\
&= \left( \int_{1}^{4} (x_1 - 1) \, dx_1 \right) \times \left( \int_{0}^{2} x_2 \, dx_2 \right) \\
&= \left[ \frac{x_1^2}{2} - x_1 \right]_{1}^{4} \times \left[ \frac{x_2^2}{2} \right]_{0}^{2} \\
&= 4.5 \times 2 = 9
\end{aligned}

\\\\

\text{JPDF}

\\\\

\int_{c}^{d} \int_{a}^{b} f(x_1,x_2) \, dx_1 \, dx_2 = 1

\\\\

\text{JPDF with Normalizing Constant}

\\\\

\int_{0}^{2} \int_{1}^{4} \frac{(x_1 - 1)x_2}{C} \, dx_1 \, dx_2 = 1

\\\\

\text{Verifying the JPDF Natively Integrates to 1 Using the Normalizing Constant:} \\

\\\\

\begin{aligned}
\text{Total Volume} &= \int_{0}^{2} \int_{1}^{4} \frac{(x_1 - 1)x_2}{9} \, dx_1 \, dx_2 \\
&= \frac{1}{9} \left( \int_{1}^{4} (x_1 - 1) \, dx_1 \right) \times \left( \int_{0}^{2} x_2 \, dx_2 \right) \\
&= \frac{1}{9} \left[ \frac{x_1^2}{2} - x_1 \right]_{1}^{4} \times \left[ \frac{x_2^2}{2} \right]_{0}^{2} \\
&= \frac{1}{9} \left( 4.5 \right) \times \left( 2 \right) \\
&= \frac{9}{9} = 1
\end{aligned}
